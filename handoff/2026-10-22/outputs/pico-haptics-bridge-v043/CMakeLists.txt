cmake_minimum_required(VERSION 3.20)
set(PICO_BOARD pico2_w CACHE STRING "Known RP2350 USB only")
set(PICO_PLATFORM rp2350-arm-s CACHE STRING "RP2350 ARM")
set(PICO_NO_FLASH 1 CACHE BOOL "Default RAM dry run")
set(PICO_FLASH_SIZE_BYTES 4194304 CACHE STRING "Verified physical Flash size")
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(pico_pro_capture C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(pico_pro_capture main.c usb_descriptors.c virtual_spi.c capture_journal.c bridge_transport.c)
option(BRIDGE_RAM_END_BOOTSEL "Diagnostic RAM session exits to BOOTSEL" OFF)
option(BRIDGE_RAM_DIAGNOSTIC_RECOVERY "Read-only RAM snapshot phase after terminal ACK" OFF)
if(BRIDGE_RAM_DIAGNOSTIC_RECOVERY)
    if(NOT PICO_NO_FLASH OR NOT BRIDGE_RAM_END_BOOTSEL)
        message(FATAL_ERROR "RAM snapshot recovery requires RAM and BOOTSEL return")
    endif()
    target_compile_definitions(pico_pro_capture PRIVATE BRIDGE_RAM_DIAGNOSTIC_RECOVERY=1)
endif()
if(BRIDGE_RAM_END_BOOTSEL)
    if(NOT PICO_NO_FLASH)
        message(FATAL_ERROR "BRIDGE_RAM_END_BOOTSEL requires RAM profile")
    endif()
    target_compile_definitions(pico_pro_capture PRIVATE BRIDGE_RAM_END_BOOTSEL=1)
endif()
set(BRIDGE_PRIVATE_DIR "${CMAKE_CURRENT_LIST_DIR}/../../work/private" CACHE PATH "Local untracked credentials directory")
target_include_directories(pico_pro_capture PRIVATE ${CMAKE_CURRENT_LIST_DIR} ${BRIDGE_PRIVATE_DIR})
target_link_libraries(pico_pro_capture pico_stdlib hardware_watchdog hardware_flash pico_flash tinyusb_device pico_cyw43_arch_lwip_poll pico_rand)
# Core1 is never launched by this firmware.
target_compile_definitions(pico_pro_capture PRIVATE PICO_FLASH_ASSUME_CORE1_SAFE=1)
pico_enable_stdio_usb(pico_pro_capture 0)
pico_enable_stdio_uart(pico_pro_capture 0)
target_compile_options(pico_pro_capture PRIVATE -Wall -Wextra -Werror)
if(PICO_NO_FLASH)
    pico_set_program_name(pico_pro_capture "bridge-pro-capture-ram-dryrun")
else()
    pico_set_program_name(pico_pro_capture "bridge-pro-capture-flash")
endif()
pico_set_program_version(pico_pro_capture "0.4.3")
pico_add_extra_outputs(pico_pro_capture)
