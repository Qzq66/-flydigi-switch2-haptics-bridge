cmake_minimum_required(VERSION 3.20)
set(PICO_BOARD pico2_w CACHE STRING "CYW43 board")
set(PICO_PLATFORM rp2350-arm-s CACHE STRING "RP2350 ARM")
set(PICO_NO_FLASH 1 CACHE BOOL "RAM only")
set(PICO_NO_PICOTOOL 1 CACHE BOOL "No tool downloads" FORCE)
set(FETCHCONTENT_FULLY_DISCONNECTED ON CACHE BOOL "Local dependencies" FORCE)
set(PICO_BTSTACK_PATH "${CMAKE_CURRENT_LIST_DIR}/../deps/btstack-eb0bb8b" CACHE PATH "Pinned BTstack")
include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(switch_input_probe C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(switch_input_probe main.c ../bt_hid_classic/ram_bank.c)
target_include_directories(switch_input_probe PRIVATE ${CMAKE_CURRENT_LIST_DIR} ../bt_hid_classic)
target_link_libraries(switch_input_probe pico_stdlib pico_btstack_classic pico_btstack_cyw43 pico_cyw43_arch_poll)
target_link_options(switch_input_probe PRIVATE "-Wl,--wrap=pico_flash_bank_instance")
target_compile_definitions(switch_input_probe PRIVATE CYW43_LWIP=0 PICO_STDIO_USB_ENABLE_RESET_VIA_BAUD_RATE=0 PICO_STDIO_USB_ENABLE_RESET_VIA_VENDOR_INTERFACE=0 PICO_STDIO_USB_CONNECT_WAIT_TIMEOUT_MS=0 PICO_STDIO_USB_DEFAULT_CRLF=0 PICO_STDIO_USB_PRODUCT_STR="Switch_Input_Probe")
set_source_files_properties(main.c PROPERTIES COMPILE_OPTIONS "-Wall;-Wextra;-Werror")
pico_enable_stdio_usb(switch_input_probe 1)
pico_enable_stdio_uart(switch_input_probe 0)
pico_set_program_name(switch_input_probe "switch-input-no-output")
pico_set_program_version(switch_input_probe "0.1.0")
