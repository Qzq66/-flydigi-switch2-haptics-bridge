cmake_minimum_required(VERSION 3.20)
set(PICO_BOARD pico2_w CACHE STRING "Known CYW43 board target")
set(PICO_PLATFORM rp2350-arm-s CACHE STRING "RP2350 ARM")
set(PICO_NO_FLASH 1 CACHE BOOL "RAM-first diagnostic image")
set(PICO_FLASH_SIZE_BYTES 4194304 CACHE STRING "Verified board capacity")
set(PICO_NO_PICOTOOL 1 CACHE BOOL "No automatic image-tool downloads" FORCE)
set(FETCHCONTENT_FULLY_DISCONNECTED ON CACHE BOOL "Local dependencies only" FORCE)
set(PICO_BTSTACK_PATH "${CMAKE_CURRENT_LIST_DIR}/../deps/btstack-eb0bb8b" CACHE PATH "Isolated SDK-pinned BTstack")
include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(flydigi_bt_hid_classic C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(flydigi_bt_hid_classic main.c ram_bank.c security_trace.c)
target_include_directories(flydigi_bt_hid_classic PRIVATE ${CMAKE_CURRENT_LIST_DIR})
target_link_libraries(flydigi_bt_hid_classic pico_stdlib pico_btstack_classic pico_btstack_cyw43 pico_cyw43_arch_poll)
target_link_options(flydigi_bt_hid_classic PRIVATE "-Wl,--wrap=pico_flash_bank_instance")
target_compile_definitions(flydigi_bt_hid_classic PRIVATE CYW43_LWIP=0 PICO_STDIO_USB_ENABLE_RESET_VIA_BAUD_RATE=0 PICO_STDIO_USB_ENABLE_RESET_VIA_VENDOR_INTERFACE=0 PICO_STDIO_USB_CONNECT_WAIT_TIMEOUT_MS=0 PICO_STDIO_USB_DEFAULT_CRLF=0 PICO_STDIO_USB_PRODUCT_STR="Flydigi_BT_Classic")
set_source_files_properties(main.c ram_bank.c security_trace.c PROPERTIES COMPILE_OPTIONS "-Wall;-Wextra;-Werror")
pico_enable_stdio_usb(flydigi_bt_hid_classic 1)
pico_enable_stdio_uart(flydigi_bt_hid_classic 0)
pico_set_program_name(flydigi_bt_hid_classic "flydigi-bt-hid-classic")
pico_set_program_version(flydigi_bt_hid_classic "0.1.0")
# No UF2 output or loader target; trial uses picotool RAM load only.
